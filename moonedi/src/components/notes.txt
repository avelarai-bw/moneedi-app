import { useState, useEffect } from 'react'; // Added useEffect for scroll
import styles from './NavBar.module.css';
import logo from '../assets/moonedi-logo-w-bg.png';
import { Link } from 'react-router-dom';
import { FaBars, FaTimes, FaUser, FaShoppingCart, FaRobot } from 'react-icons/fa'; // Added icons for actions
import { motion } from 'framer-motion'; // For elegant dropdowns

const NavBar = () => {
  const [isOpen, setIsOpen] = useState(false);
  const [isScrolled, setIsScrolled] = useState(false); // New: For sticky shrink

  // New: Scroll listener for elegant sticky effect
  useEffect(() => {
    const handleScroll = () => setIsScrolled(window.scrollY > 50);
    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, []);

  const toggleMenu = () => setIsOpen(!isOpen);
  const closeMenu = () => setIsOpen(false);

  return (
    <motion.header // Wrap in motion for global entrance
      className={`${styles.navbar} ${isScrolled ? styles.scrolled : ''}`}
      initial={{ y: -100 }}
      animate={{ y: 0 }}
      transition={{ duration: 0.6 }}
    >
      <div className={styles.navbar__logo}>
        <Link to="/">
          <img src={logo} alt="Moonedi Enterprise" />
        </Link>
      </div>

      <nav className={styles.navbar__nav}>
        <ul className={styles.navList}>
          <li><Link to="/" onClick={closeMenu}>Home</Link></li>
          <li className={styles.dropdown}>
            <Link to="/about" onClick={closeMenu}>About</Link>
            <motion.ul // Motion for smooth dropdown
              className={styles.dropdownMenu}
              initial={{ opacity: 0, scale: 0.95, y: -10 }}
              whileHover={{ opacity: 1, scale: 1, y: 0 }}
              transition={{ duration: 0.2 }}
            >
              <li><Link to="/abattoir" onClick={closeMenu}>Abattoir</Link></li>
              <li><Link to="/meat-processing" onClick={closeMenu}>Meat Processing</Link></li>
              <li><Link to="/distribution" onClick={closeMenu}>Distribution</Link></li>
            </motion.ul>
          </li>
          <li className={styles.dropdown}>
            <Link to="/customers" onClick={closeMenu}>Our Customers</Link>
            <motion.ul
              className={styles.dropdownMenu}
              initial={{ opacity: 0, scale: 0.95, y: -10 }}
              whileHover={{ opacity: 1, scale: 1, y: 0 }}
              transition={{ duration: 0.2 }}
            >
              <li><Link to="/customers/botswana" onClick={closeMenu}>Botswana</Link></li>
              <li><Link to="/customers/sadc" onClick={closeMenu}>SADC Region</Link></li>
              <li><Link to="/customers/europe" onClick={closeMenu}>Europe</Link></li> {/* Highlighted global */}
            </motion.ul>
          </li>
          <li className={styles.dropdown}>
            <Link to="/standards" onClick={closeMenu}>Standards</Link> {/* Merged Health + Quality */}
            <motion.ul
              className={styles.dropdownMenu}
              initial={{ opacity: 0, scale: 0.95, y: -10 }}
              whileHover={{ opacity: 1, scale: 1, y: 0 }}
              transition={{ duration: 0.2 }}
            >
              <li><Link to="/health/hygiene-policy" onClick={closeMenu}>Hygiene & Safety</Link></li>
              <li><Link to="/quality/policy" onClick={closeMenu}>Quality Policies</Link></li>
              <li><Link to="/quality/certificate" onClick={closeMenu}>Certifications</Link></li>
            </motion.ul>
          </li>
          <li><Link to="/product-gallery" onClick={closeMenu}>Health & Hygiene</Link></li> {/* Shortened name */}
        </ul>
      </nav>

      {/* Updated: Actions as icon buttons—no desktop hamburger */}
      <div className={styles.navbar__actions}>
       
        <Link to="/order-online" className={styles.actionIcon} title="Place Order" onClick={closeMenu}>
          <FaShoppingCart />
        </Link>
        <Link to="/customer-login" className={styles.actionIcon} title="Login" onClick={closeMenu}>
          <FaUser />
        </Link>
        <button className={styles.askAI__button}>
          <FaRobot /> AI
        </button>
        <button className={styles.hamburger} onClick={toggleMenu}>
          {isOpen ? <FaTimes /> : <FaBars />}
        </button>
      </div>

      {isOpen && (
        <motion.div // Motion for mobile slide
          className={styles.mobileMenu}
          initial={{ x: '100%' }}
          animate={{ x: 0 }}
          exit={{ x: '100%' }}
          transition={{ duration: 0.4 }}
        >
          <ul className={styles.mobileNavList}>
            {/* Mirror desktop structure, but with accordion logic if you add state */}
            <li><Link to="/" className={styles.mobileMainLink} onClick={closeMenu}>Home</Link></li>
            <li>
              <Link to="/about" className={styles.mobileMainLink} onClick={closeMenu}>About</Link>
              <ul className={styles.mobileSubMenu}>
                <li><Link to="/abattoir" onClick={closeMenu}>Abattoir</Link></li>
                <li><Link to="/meat-processing" onClick={closeMenu}>Meat Processing</Link></li>
                <li><Link to="/distribution" onClick={closeMenu}>Distribution</Link></li>
              </ul>
            </li>
            {/* ... Similar for other mains, with shortened subs ... */}
            <li className={styles.mobileDivider} />
            <li><Link to="/order-online" className={styles.mobileActionLink} onClick={closeMenu}>Order Online</Link></li>
            <li><Link to="/quote-request" className={styles.mobileActionLink} onClick={closeMenu}>Get Quote</Link></li>
            <li><Link to="/customer-login" className={styles.mobileActionLink} onClick={closeMenu}>Login</Link></li>
          </ul>
        </motion.div>
      )}

      {isOpen && <div className={styles.overlay} onClick={closeMenu} />}
    </motion.header>
  );
};
